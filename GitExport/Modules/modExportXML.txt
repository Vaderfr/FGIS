Option Compare Database
Option Explicit
    Private DataStatusId As Integer
    Private mDB As DAO.Database
    

Public Sub ExportProtocols()
On Error GoTo Er

    Dim doc As Object
    Dim Root As Object
    Dim rs As DAO.Recordset
    Dim StatusId As VbMsgBoxResult
    Dim MaxProtocols As Long
    Dim FilePath As String
    Dim FileName As String
    Dim fso As Scripting.FileSystemObject
    
    Dim CountInFile As Long
    Dim ProtocolNum As Long
    Dim FirstNum As Long
    Dim LastNum As Long

    MaxProtocols = fGetSetting("Paсk")
    If MaxProtocols <= 0 Then MaxProtocols = 20
    
    FilePath = fGetSetting("Destination")

    Set mDB = CurrentDb
    
    If Not CheckProtocolScans Then GoTo Exit_Sub
    
    Set rs = mDB.OpenRecordset("SELECT p.* " _
        & "FROM (Protocol AS p INNER JOIN ProtocolScan AS s " _
        & "ON p.IdProtocol = s.IdOProtocol) LEFT JOIN tProtCab ON p.DocId = tProtCab.ProtocolNumber " _
        & "WHERE (((tProtCab.ProtocolNumber) Is Null));")
        
    If rs.EOF Then MsgBox "Нет на выгрузку...", vbOKOnly + vbExclamation: GoTo Exit_Sub

    StatusId = MsgBox("Чистовик?", vbYesNo + vbQuestion)

    If StatusId = vbYes Then
        DataStatusId = 13
    Else
        DataStatusId = 20
    End If

    Set fso = New Scripting.FileSystemObject

    ProtocolNum = 0
    CountInFile = 0

    CreateXml doc, Root


    Do Until rs.EOF

        ProtocolNum = ProtocolNum + 1
        CountInFile = CountInFile + 1

        WriteProtocol doc, Root, rs


        If CountInFile = MaxProtocols Then

            FirstNum = ProtocolNum - CountInFile + 1
            LastNum = ProtocolNum

            FileName = fso.BuildPath( _
                FilePath, _
                "xml_" & FirstNum & "-" & LastNum & ".xml")

            doc.Save FileName
            
            Set Root = Nothing
            Set doc = Nothing

            CountInFile = 0
            CreateXml doc, Root

        End If


        rs.MoveNext

    Loop


    If CountInFile > 0 Then

        FirstNum = ProtocolNum - CountInFile + 1
        LastNum = ProtocolNum

        FileName = fso.BuildPath( _
            FilePath, _
            "xml_" & FirstNum & "-" & LastNum & ".xml")

        doc.Save FileName

    End If

Exit_Sub:

    If Not rs Is Nothing Then
        rs.Close
        Set rs = Nothing
    End If

    Set mDB = Nothing
    Set Root = Nothing
    Set doc = Nothing
    Set fso = Nothing
    
    Exit Sub

Er:
    If ErrMsg("modExportXML.ExportProtocols") Then
        Resume
    Else
        Resume Exit_Sub
    End If
End Sub

Private Sub CreateXml(ByRef doc As Object, ByRef Root As Object)
On Error GoTo Er

    Set doc = XmlCreate()
    Set Root = XmlNode(doc, "root")
    XmlAppend doc, Root

Exit_Sub:

    Exit Sub

Er:
    If ErrMsg("modExportXML.CreateXml") Then
        Resume
    Else
        Resume Exit_Sub
    End If
End Sub


Private Sub WriteProtocol(doc As Object, Root As Object, rs As DAO.Recordset)
On Error GoTo Er

    Dim p As Object
    Dim finalStatus As Long

    If IsReplacementProtocol(rs!docId) Then
        finalStatus = 20
    Else
        finalStatus = DataStatusId
    End If
    
    
    Set p = XmlNode(doc, "protocol")
    XmlAppend Root, p

    ' === обязательные поля (XSD sequence порядок важен) ===
    XmlText doc, p, "DocId", rs!docId
    XmlDate doc, p, "DocCreationDate", rs!DocCreationDate
    XmlDate doc, p, "DocStartDate", rs!DocStartDate
    XmlDate doc, p, "DocValidityDate", rs!DocValidityDate


    XmlInt doc, p, "DataStatusId", finalStatus
    XmlInt doc, p, "ProtocolStatusId", rs!ProtocolStatusId

    ' === вложенные структуры дальше ===
    WriteProtocolScan doc, p, rs!IdProtocol
    WriteAddressIDs doc, p, rs!IdProtocol
    WriteTerritoryFeature doc, p, rs!TerritoryFeature
    WriteProtocolAddress doc, p, rs!IdProtocol

    XmlText doc, p, "AdditionKindId", rs!AdditionKindId
    XmlDate doc, p, "AdditionKindDate", rs!AdditionKindDate
    XmlDate doc, p, "ApplicationDate", rs!ApplicationDate

    WriteCustomer doc, p, rs!IdProtocol

    XmlBool doc, p, "AccredScope353", rs!AccredScope353
    XmlBool doc, p, "NoEquipmentInfo", rs!NoEquipmentInfo

    WriteEquipment doc, p, rs!IdProtocol
    WriteApprovedUsers doc, p, rs!IdProtocol

    WriteObjectInfo doc, p, rs!IdProtocol

Exit_Sub:

    Set p = Nothing
    
    Exit Sub

Er:
    If ErrMsg("modExportXML.WriteProtocol") Then
        Resume
    Else
        Resume Exit_Sub
    End If
End Sub


Private Function IsReplacementProtocol(docId As String) As Boolean
On Error GoTo Er

    Dim re As Object
    Set re = CreateObject("VBScript.RegExp")

    re.Pattern = "/[0-9]*И$"
    re.Global = False
    re.IgnoreCase = True

    IsReplacementProtocol = re.Test(docId)
    
Exit_Sub:
    Set re = Nothing
    Exit Function

Er:
    If ErrMsg("modExportXML.CreateXml") Then
        Resume
    Else
        Resume Exit_Sub
    End If

End Function

Public Sub WriteAnotherDocs(doc As Object, parent As Object, _
                            objectInfoId As Long)

    Dim rs As DAO.Recordset
    Dim grp As Object
    Dim det As Object

    Set rs = mDB.OpenRecordset( _
        "SELECT AnotherDocName, AnotherDocNumber " & _
        "FROM AnotherDocDetails WHERE IdOObjectInfo=" & objectInfoId)

    ' Нет записей
    If rs.EOF Then

        rs.Close

        ' Заглушка нужна только если отсутствует MethodDocId
If Not HasMethodDocs(objectInfoId) Then
            Set grp = doc.createElement("AnotherDoc")
            parent.appendChild grp

            Set det = doc.createElement("AnotherDocDetails")
            grp.appendChild det

            XmlText doc, det, "AnotherDocName", "-"
            XmlText doc, det, "AnotherDocNumber", "-"

        End If

        Exit Sub

    End If

    ' Есть реальные записи
    Set grp = doc.createElement("AnotherDoc")
    parent.appendChild grp

    Do Until rs.EOF

        Set det = doc.createElement("AnotherDocDetails")
        grp.appendChild det

        XmlText doc, det, "AnotherDocName", rs!AnotherDocName
        XmlText doc, det, "AnotherDocNumber", rs!AnotherDocNumber

        rs.MoveNext
    Loop

    rs.Close

End Sub


Public Function XmlCreate() As Object
    Dim doc As Object
    Set doc = CreateObject("MSXML2.DOMDocument.6.0")

    doc.appendChild doc.createProcessingInstruction( _
        "xml", "version='1.0' encoding='utf-8'")

    Set XmlCreate = doc
End Function

Public Function XmlNode(doc As Object, nodeName As String) As Object
    Set XmlNode = doc.createElement(nodeName)
End Function

Public Sub XmlAppend(parent As Object, child As Object)
    parent.appendChild child
End Sub

Public Function XmlHasValue(v As Variant) As Boolean

    If IsNull(v) Then
        XmlHasValue = False
        Exit Function
    End If

    If VarType(v) = vbString Then
        XmlHasValue = (Len(Trim$(v)) > 0)
    Else
        XmlHasValue = True
    End If

End Function



Public Sub XmlText(doc As Object, parent As Object, tag As String, Value As Variant)

    If Not XmlHasValue(Value) Then Exit Sub

    Dim n As Object
    Set n = doc.createElement(tag)
    n.Text = CStr(Value)

    parent.appendChild n

End Sub
Public Sub XmlInt(doc As Object, parent As Object, tag As String, Value As Variant)

    If Not XmlHasValue(Value) Then Exit Sub
    
    Dim n As Object
    Set n = doc.createElement(tag)
    n.Text = CStr(CLng(Value))

    parent.appendChild n

End Sub


Public Sub XmlBool(doc As Object, parent As Object, tag As String, Value As Variant)

    If Not XmlHasValue(Value) Then Exit Sub
    
    Dim n As Object
    Set n = doc.createElement(tag)

    If Value Then
        n.Text = "true"
    Else
        n.Text = "false"
    End If

    parent.appendChild n

End Sub


Public Sub XmlDate(doc As Object, parent As Object, tag As String, Value As Variant)

    If Not XmlHasValue(Value) Then Exit Sub

    Dim n As Object
    Set n = doc.createElement(tag)

    n.Text = Format(Value, "yyyy-mm-dd")

    parent.appendChild n

End Sub

Private Function HasAnotherDocs(objectInfoId As Long) As Boolean

    HasAnotherDocs = _
        (DCount("*", "AnotherDocDetails", _
                "IdOObjectInfo=" & objectInfoId) > 0)

End Function


Private Function NeedAnotherDocStub(objectInfoId As Long, methodDocId As Variant) As Boolean

    If Not IsNull(methodDocId) Then
        NeedAnotherDocStub = False
    Else
        NeedAnotherDocStub = Not HasAnotherDocs(objectInfoId)
    End If

End Function
Public Sub XmlFloat(doc As Object, parent As Object, tag As String, Value As Variant)

    If Not XmlHasValue(Value) Then Exit Sub

    Dim n As Object
    Set n = doc.createElement(tag)

    n.Text = Replace(Trim$(Str$(CSng(Value))), ",", ".")

    parent.appendChild n

End Sub

Private Function HasMethodDocs(objectInfoId As Long) As Boolean

    HasMethodDocs = _
        (DCount("*", "MethodDocId", _
                "IdOObjectInfo=" & objectInfoId) > 0)

End Function
Private Function NeedIsAnotherDoc(objectInfoId As Long) As Boolean

    If HasAnotherDocs(objectInfoId) Then
        NeedIsAnotherDoc = True

    ElseIf HasMethodDocs(objectInfoId) Then
        NeedIsAnotherDoc = False

    Else
        NeedIsAnotherDoc = True
    End If

End Function


Private Function CheckProtocolScans() As Boolean

    Dim rs As DAO.Recordset
    Dim Msg As String

    Set rs = mDB.OpenRecordset( _
        "SELECT p.DocId, s.FileName " & _
        "FROM Protocol AS p INNER JOIN ProtocolScan AS s " & _
        "ON p.IdProtocol = s.IdOProtocol " & _
        "WHERE s.IdOProtocol IN (" & _
        "SELECT IdOProtocol FROM ProtocolScan " & _
        "GROUP BY IdOProtocol HAVING Count(*) > 1) " & _
        "ORDER BY p.DocId, s.FileName")

    If Not rs.EOF Then

        Msg = "Обнаружены протоколы с несколькими сканами:" & vbCrLf & vbCrLf

        Do Until rs.EOF

            Msg = Msg & _
                rs!docId & " : " & rs!FileName & vbCrLf

            rs.MoveNext

        Loop

        MsgBox Msg, vbExclamation, "Ошибка сканов"

        CheckProtocolScans = False

    Else

        CheckProtocolScans = True

    End If

    rs.Close
    Set rs = Nothing

End Function
