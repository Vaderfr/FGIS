Const PAGE_SIZE As Long = 500

Public Sub sProtLoad()
    On Error GoTo Er

    Dim rsProt As DAO.Recordset
    Dim json As Object
    Dim items As Object
    Dim itm As Object
    
    Dim sAuthorization As String
    Dim sOrgId As String
    Dim sURL As String
    Dim sJSON As String
    Dim dtDate As String
    
    Dim iDayFgis As Long
    Dim iOffset As Long
    Dim currentPageCount As Long

    iDayFgis = fGetSetting("DayFgis")
    dtDate = Format(DateAdd("d", -iDayFgis, Date), "yyyy-mm-dd")
    sOrgId = fGetSetting("OrgId")

    ' 1. Получение или запрос токена (исправлена опечатка: "Authorization")
    If GetSetting("3EIAS", "Authorization", "Token", "") = "" Then
        SaveSetting "3EIAS", "Authorization", "Token", fInputBox("Введите токен безопасности")
    End If
    
    sAuthorization = "Bearer " & GetSetting("3EIAS", "Authorization", "Token", "")

    sURL = "https://fgis.fsa.gov.ru/api/v1/rpi/common/showcases/get"
'Debug.Print "DELETE FROM tProtCab " _
        & "WHERE CreateDate >= #" & dtDate & "#;"
    ' 2. Очистка старых записей
    CurrentDb.Execute "DELETE FROM tProtCab " _
        & "WHERE CreateDate >= #" & dtDate & "#;" ', dbFailOnError

    Set rsProt = CurrentDb.OpenRecordset("tProtCab", dbOpenDynaset)

    Do
        sJSON = fPostJSON(sURL, fProtPayload(dtDate, iOffset), sAuthorization, sOrgId)
        
        ' 3. Парсинг JSON нативными средствами VBA (совместимо с 64-бит)
        Set json = JsonConverter.ParseJson(sJSON)
        
        If Not json.Exists("items") Or TypeName(json("items")) <> "Collection" Then
            Err.Raise vbObjectError + 1000, , "Неверный формат ответа API: " & Left$(sJSON, 200)
        End If
        
        Set items = json("items")
        currentPageCount = items.count
        
        ' 4. Добавление записей в таблицу
        

For Each itm In items
    rsProt.AddNew
    rsProt!id = itm("id")
    rsProt!Edit = itm("edit")
    rsProt!Delay = Nz(itm("delay"), "")
    rsProt!ALName = itm("ALName")
    rsProt!SentDate = ISOToDate(itm("sentDate"))
    rsProt!CreateDate = ISOToDate(itm("createDate"))
    rsProt!ObjectType = itm("objectType")
    rsProt!RegNumberAL = itm("regNumberAL")
    rsProt!CustomerName = itm("customerName")
    rsProt!CustomerType = itm("customerType")
    rsProt!DecisionDate = itm("decisionDate")
    rsProt!ProtocolDate = itm("protocolDate")
    rsProt!StatementDate = itm("statementDate")
    rsProt!AccredScope353 = itm("accredScope353")
    rsProt!FullObjectName = itm("fullObjectName")
    rsProt!ProtocolNumber = itm("protocolNumber")
    rsProt!ProtocolStatus = itm("protocolStatus")
    rsProt!ResearchResult = itm("researchResult")
    rsProt!ChangeStatusNumber = itm("changeStatusNumber")
    rsProt!ProtocolDataStatus = itm("protocolDataStatus")
    rsProt.Update
Next
'        For Each itm In items
'            rsProt.AddNew
'            rsProt![Номер протокола] = itm("protocolNumber")
'            rsProt![Дата протокола] = itm("protocolDate")
'            rsProt.Update
'        Next itm
        
        iOffset = iOffset + 1
        DoEvents

    ' 5. ИСПРАВЛЕНО: Цикл продолжается только если на текущей странице вернулось полное число записей (PAGE_SIZE)
    Loop While currentPageCount = PAGE_SIZE

Exit_Sub:
    On Error Resume Next
    If Not rsProt Is Nothing Then
        rsProt.Close
        Set rsProt = Nothing
    End If
    Set items = Nothing
    Set json = Nothing
    Exit Sub

Er:
    If ErrMsg("bHTTP.sProtLoad") Then
        Resume
    Else
        Resume Exit_Sub
    End If
End Sub

Private Function fPostJSON(ByVal sURL As String, _
                           ByVal sPayload As String, _
                           ByRef sAuthorization As String, _
                           ByVal sOrgId As String) As String
    On Error GoTo Er

    ' Используем позднее связывание для лучшей совместимости
    Dim http As Object
    Set http = CreateObject("MSXML2.XMLHTTP.6.0")

Retry:
    With http
        .Open "POST", sURL, False
        .setRequestHeader "Accept", "application/json, text/plain, */*"
        .setRequestHeader "Authorization", sAuthorization
        .setRequestHeader "Cookie", "session_lk=" & Replace$(sAuthorization, "Bearer ", "")
        .setRequestHeader "Content-Type", "application/json"
        .setRequestHeader "lkid", "8"
        .setRequestHeader "orgid", "orgs.dict.orgtypes.il." & sOrgId
        .setRequestHeader "Origin", "https://fgis.fsa.gov.ru"
        .setRequestHeader "Referer", "https://fgis.fsa.gov.ru/rpi"
        .setRequestHeader "User-Agent", "Mozilla/5.0 (Windows NT 10.0; Win64; x64)"
        .send sPayload
    End With

    Select Case http.Status
        Case 200
            fPostJSON = http.responseText
            
        Case 401
            Dim newToken As String
            newToken = fInputBox("Срок действия токена истек. Введите новый токен безопасности:")
            
            ' Защита от бесконечного цикла: если пользователь нажал Отмена или оставил поле пустым
            If Len(Trim$(newToken)) = 0 Then
                Err.Raise vbObjectError + 401, , "Операция отменена: токен не введен."
            End If
            
            SaveSetting "3EIAS", "Authorization", "Token", newToken
            sAuthorization = "Bearer " & newToken
            GoTo Retry
            
        Case Else
            Err.Raise vbObjectError + http.Status, , "HTTP " & http.Status & vbCrLf & http.responseText
    End Select

Exit_Function:
    Set http = Nothing
    Exit Function

Er:
    ' Гарантированная очистка объекта http даже при ошибке
    Resume Exit_Function
End Function

        """hardwareNumber"":null," & _
        """ALRegisterNumber"":null," & _
        """ALName"":null," & _
        """objectType"":null," & _

        """statementDate"":null," & _
        """clientType"":null," & _

End Function

Private Function GetMinProtDate(sFile As String) As Date

    Dim cn As Object
    Dim rs As Object

    Set cn = CreateObject("ADODB.Connection")

    cn.Open "Provider=Microsoft.ACE.OLEDB.12.0;" & _
            "Data Source=" & sFile & ";" & _
            "Extended Properties=""Excel 12.0 Xml;HDR=YES"""

    Set rs = cn.Execute( _
        "SELECT MIN([Дата протокола]) AS MinDate FROM [registry$]" _
    )

    If Not rs.EOF Then
        GetMinProtDate = rs!MinDate
    End If

    rs.Close
    cn.Close

End Function
